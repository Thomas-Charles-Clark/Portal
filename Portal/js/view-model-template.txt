const viewModel = ((resources) => {
    let self = {
        resources: resources
    };
    
    //#region Properties
    
    // ELI5 it...

    // Are we busy?
    self.busy = ko.observable(false);

    //#endregion


    //#region Components



    //#endregion


    //#region Subscriptions



    //#endregion


    //#region Computeds



    //#endregion


    //#region Methods
    
    // To be called when we are first ready
    self.ready = () => {
        self.busy(false);
    }

    // An example of a Fetch POST call
    self.fetch = () => {
        fetch('/api/...', {
            method: 'POST',
            headers: window.fetchHeaders,
            cache: 'no-store',
            credentials: 'same-origin',
            body: self.toJSON()
        })
        .then(window.handleFetchResponse)
        .then(data => {
            console.trace(data);
        })
        .catch(err => window.catchFetchError)
    }

    // We often want to serialize our salient properties to JS
    self.toJS = () => {
        return {};
    }

    // We often want to stringify our salient properties
    self.toJSON = () => JSON.stringify(self.toJS());

    //#endregion

    return self;
});

// demonstrates a factory function with parameter destructuring and defaults
const demoFactoryFunction = ({
    userName = 'Unknown',
    avatar = 'anon.png'
} = {}) => ({
    userName,
    avatar
});
console.log(
    // { purpose: "echo", image: 'anon.png' }
    demoFactoryFunction({ userName: 'echo' }),

    // { purpose: "Anonymous", image: 'anon.png' }
    demoFactoryFunction()
);
