		let formattingCulture = e[1][0].FormattingCulture;
		let currencyCode = e[1][0].CurrencyCode;

		let totals = {
			tradeAccountCount: formatNumber(e[1].reduce((accumulator, currentItem) => accumulator += currentItem.TradeAccountCount, 0), 0, formattingCulture),
			creditStoppedCount: formatNumber(e[1].reduce((accumulator, currentItem) => accumulator += currentItem.CreditStoppedCount, 0), 0, formattingCulture),
			twelveMonthOrderCount: formatNumber(e[1].reduce((accumulator, currentItem) => accumulator += currentItem.TwelveMonthOrderCount, 0), 0, formattingCulture),
			twelveMonthOrderValue: formatCurrency(e[1].reduce((accumulator, currentItem) => accumulator += currentItem.TwelveMonthOrderValue, 0), 0, formattingCulture, currencyCode)
		};
		e.splice(1, 0, totals);
		return e;
	});

	let records = Map.groupBy(items.flatMap(i => i[2]), item => item.PeriodLabel).entries().toArray();

		currencyCode: cg[0],

		tradeAccountCount: records.flatMap(m => {
